cmake_minimum_required(VERSION 3.18)
project(pfe_native LANGUAGES CXX)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(_native cpp/native.cpp)
target_compile_features(_native PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(_native PRIVATE /O2 /fp:precise)
else()
  target_compile_options(_native PRIVATE -O3 -fno-fast-math -ffp-contract=off -Wall -Wextra)
endif()
install(TARGETS _native LIBRARY DESTINATION pfe RUNTIME DESTINATION pfe)
